name: Fenix scan (market close)
on:
  schedule:
    - cron: '20 10 * * 1-5'   # 10:20 UTC = 3:50 PM IST, Mon-Fri
  workflow_dispatch: {}        # haath se bhi chala sakte ho
permissions:
  contents: write
jobs:
  scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Scan + Telegram
        run: node scan.mjs
        env:
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
      - name: Result save karo (forward-test log)
        run: |
          git config user.name "fenix-bot"
          git config user.email "fenix-bot@users.noreply.github.com"
          git add data
          git diff --staged --quiet || (git commit -m "scan $(date -u +%F)" && git push)
